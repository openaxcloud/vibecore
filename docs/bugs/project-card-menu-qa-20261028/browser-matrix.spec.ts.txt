import {test, expect, chromium, type Page, type APIRequestContext, type Locator} from '/workspace/vibecore/node_modules/@playwright/test/index.mjs';
import {writeFileSync} from 'node:fs';
const web='http://127.0.0.1:5199',api='http://127.0.0.1:3018';
const longName='Projet QA — '+ 'nom-tres-long-'.repeat(10);
async function fixture(page:Page,request:APIRequestContext){
 const signup=await request.post(api+'/auth/register',{data:{email:`project-menu-${Date.now()}-${Math.random().toString(36).slice(2)}@local.test`,password:'Password123!',name:'Isolated menu QA'}});expect(signup.ok()).toBe(true);const user=await signup.json();const headers={authorization:`Bearer ${user.token}`};
 const created=await request.post(`${api}/orgs/${user.organization.id}/projects`,{headers,data:{name:longName}});expect(created.ok()).toBe(true);const project=(await created.json()).project;
 await page.context().addCookies([{name:'vc_session',value:user.token,url:web,httpOnly:true,sameSite:'Lax'}]);
 const mutations:string[]=[];let handleMutation:undefined|((route:any)=>Promise<void>);
 await page.route('**/*',async route=>{const req=route.request(),u=new URL(req.url());if(!['localhost','127.0.0.1'].includes(u.hostname)){await route.abort();return;}if(!['GET','HEAD','OPTIONS'].includes(req.method())){mutations.push(u.pathname);if(handleMutation&&u.pathname.replace(/\.data$/,'')===`/api/projects/${project.id}/project-action`){await handleMutation(route);return;}await route.abort();return;}await route.continue();});
 await page.goto(web+'/projects',{waitUntil:'domcontentloaded'});await expect(page.locator('html')).toHaveAttribute('data-ecode-hydrated','true');
 const trigger=page.getByRole('button',{name:new RegExp('^(Project actions for|Actions du projet) '+longName.replace(/[.*+?^${}()|[\]\\]/g,'\\$&'))});await expect(trigger).toBeVisible({timeout:60000});
 return {project,headers,trigger,mutations,setMutation:(handler:(route:any)=>Promise<void>)=>{handleMutation=handler;}};
}
async function bounds(locator:Locator,page:Page,min44=false){await locator.evaluate(async el=>{const animations=[];for(let node:Element|null=el;node;node=node.parentElement)animations.push(...node.getAnimations());await Promise.all(animations.map(a=>a.finished.catch(()=>{})));});const box=await locator.boundingBox();expect(box).not.toBeNull();const width=await page.evaluate(()=>innerWidth);expect(box!.x).toBeGreaterThanOrEqual(-1);expect(box!.x+box!.width).toBeLessThanOrEqual(width+1);if(min44){expect(box!.height).toBeGreaterThanOrEqual(44);expect(box!.width).toBeGreaterThanOrEqual(44);}return box;}
async function menuProtocol(page:Page,request:APIRequestContext,info:any){
 const f=await fixture(page,request);await bounds(f.trigger,page,true);await f.trigger.focus();await page.keyboard.press('Enter');const menu=page.getByRole('menu');await expect(menu).toBeVisible();await bounds(menu,page);for(const item of await menu.getByRole('menuitem').all())await bounds(item,page,true);await page.keyboard.press('ArrowDown');await page.keyboard.press('Escape');await expect(menu).not.toBeVisible();await expect(f.trigger).toBeFocused();
 await f.trigger.click();await page.getByRole('menuitem',{name:/^(Rename|Renommer)$/}).click();const input=page.getByRole('textbox',{name:/^(Rename project|Renommer le projet)/});await expect(input).toBeVisible();await input.fill('draft that must not be saved');await input.press('Escape');await expect(input).toHaveCount(0);await page.keyboard.press('Tab');expect(f.mutations.filter(path=>!/^\/api\/user\/preferences(?:\.data)?$/.test(path))).toEqual([]);
 await f.trigger.click();await page.getByRole('menuitem',{name:/^(Delete|Supprimer)$/}).click();const dialog=page.getByRole('dialog');await expect(dialog).toBeVisible();await bounds(dialog,page);await bounds(dialog.getByRole('button',{name:/^(Delete project|Supprimer le projet)$/}),page,true);await bounds(dialog.getByRole('button',{name:/^(Cancel|Annuler)$/}),page,true);await page.screenshot({path:info.outputPath('delete-long-name.png')});await dialog.getByRole('button',{name:/^(Cancel|Annuler)$/}).click();await expect(dialog).not.toBeVisible();expect(f.mutations.filter(path=>!/^\/api\/user\/preferences(?:\.data)?$/.test(path))).toEqual([]);
 await f.trigger.click();await page.getByRole('menuitem',{name:/^(Delete|Supprimer)$/}).click();await expect(dialog).toBeVisible();await page.keyboard.press('Escape');await expect(dialog).not.toBeVisible();expect(f.mutations.filter(path=>!/^\/api\/user\/preferences(?:\.data)?$/.test(path))).toEqual([]);await page.screenshot({path:info.outputPath('project-menu.png')});writeFileSync(info.outputPath('blocked-mutations.json'),JSON.stringify(f.mutations,null,2));
}
for(const width of [320,390,768,1024,1440])test(`menu layout keyboard cancel ${width}`,async({page,request},info)=>{test.setTimeout(120000);await page.setViewportSize({width,height:900});await menuProtocol(page,request,info);});
for(const width of [320,1440])test(`rename controlled error then real persistence ${width}`,async({page,request},info)=>{test.setTimeout(120000);await page.setViewportSize({width,height:900});const f=await fixture(page,request);let calls=0;let release!:()=>void;const gate=new Promise<void>(resolve=>{release=resolve;});f.setMutation(async route=>{calls++;const body=route.request().postData()??'';expect(body).toContain('rename');if(calls===1){await gate;await route.continue({postData:new URLSearchParams({intent:'rename',name:''}).toString(),headers:{...route.request().headers(),'content-type':'application/x-www-form-urlencoded'}});}else if(calls===2){await route.continue();}else{await route.abort();}});
 await f.trigger.click();await page.getByRole('menuitem',{name:/^(Rename|Renommer)$/}).click();const input=page.getByRole('textbox',{name:/^(Rename project|Renommer le projet)/});const next='Renamed isolated fixture '+width;await input.fill(next);await input.press('Enter');await input.evaluate(el=>(el as HTMLInputElement).blur());await expect(input).toBeDisabled();expect(calls).toBe(1);release();await expect(input).toBeEnabled();await expect(input).toHaveValue(next);await expect(page.getByText(/(could not be renamed|Impossible de renommer)/)).toBeVisible();const unchanged=await request.get(`${api}/projects/${f.project.id}`,{headers:f.headers});expect(unchanged.ok()).toBe(true);expect((await unchanged.json()).project.name).toBe(longName);await page.screenshot({path:info.outputPath('rename-error-draft.png')});await input.press('Enter');await expect(input).toHaveCount(0);expect(calls).toBe(2);const result=await request.get(`${api}/projects/${f.project.id}`,{headers:f.headers});expect(result.ok()).toBe(true);expect((await result.json()).project.name).toBe(next);await page.screenshot({path:info.outputPath('rename-real-persisted.png')});});

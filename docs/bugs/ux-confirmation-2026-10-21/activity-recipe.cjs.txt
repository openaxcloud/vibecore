const {chromium,expect}=require('@playwright/test');const fs=require('node:fs');
(async()=>{const s=JSON.parse(fs.readFileSync('.vibecore/ux-qa/session.json'));const b=await chromium.launch({executablePath:'/usr/bin/chromium'});const results=[];
for(const [width,height] of [[1440,900],[768,1024],[390,844]]) {const c=await b.newContext({viewport:{width,height},hasTouch:width<1200,isMobile:width===390});await c.addCookies(s.cookies);const p=await c.newPage();await p.goto(`http://127.0.0.1:5197/projects/${s.projectId}/ide?panel=activity&lang=fr`);const panel=p.locator('[data-testid="ide-service-panel"][data-panel="activity"]');const search=panel.locator('.bolt-project-activity-filters input');await search.waitFor();const rows=panel.locator('.bolt-project-activity-event');await rows.first().waitFor();await expect(rows.first()).toBeVisible();const initial=await rows.count();await search.fill('NO_MATCH_SYNTHETIC_UX_20261021');await expect(rows).toHaveCount(0);const emptyText=await panel.locator('.bolt-project-activity-list').innerText();await p.screenshot({path:`docs/bugs/ux-confirmation-2026-10-21/activity-empty-${width}.png`});await search.fill('');await expect(rows.first()).toBeVisible();const entry=rows.first().locator('button').first();await entry.focus();await p.keyboard.press('Enter');await expect(entry).toHaveAttribute('aria-expanded','true');await expect(rows.first().locator('pre')).toBeVisible();await p.keyboard.press('Enter');await expect(entry).toHaveAttribute('aria-expanded','false');await expect(rows.first().locator('pre')).toBeHidden();
await c.setOffline(true);await panel.getByRole('button',{name:/Actualiser maintenant|Refresh now/}).click();const alert=panel.getByRole('alert');await expect(alert).toBeVisible();const errorText=await alert.innerText();await p.screenshot({path:`docs/bugs/ux-confirmation-2026-10-21/activity-offline-${width}.png`});await c.setOffline(false);const recovery=p.waitForResponse(r=>r.request().method()==='GET' && r.url().includes('/ide-panel/activity') && r.status()===200);await alert.getByRole('button').click();await recovery;await expect(alert).toBeHidden();await rows.first().waitFor();await expect(rows.first()).toBeVisible();const overflow=await p.evaluate(()=>document.documentElement.scrollWidth>innerWidth);expect(overflow).toBe(false);await p.screenshot({path:`docs/bugs/ux-confirmation-2026-10-21/activity-recovered-${width}.png`});results.push({width,height,initial,emptyText,errorText,searchClear:true,keyboardExpandCollapse:true,offlineRecovery:true,horizontalOverflow:false});await c.close();}
fs.writeFileSync('docs/bugs/ux-confirmation-2026-10-21/activity.json',JSON.stringify(results,null,2));console.log(JSON.stringify(results,null,2));await b.close();})();

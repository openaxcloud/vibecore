const{chromium,expect}=require('@playwright/test');const fs=require('node:fs');
(async()=>{const {cases}=JSON.parse(fs.readFileSync('.vibecore/ux-qa/snapshot-sessions.json'));const b=await chromium.launch({executablePath:'/usr/bin/chromium'});const results=[];
for(const [format,width,height] of [['desktop',1440,900],['tablet',768,1024],['mobile',390,844]]) for(const language of ['en','fr']) {
const s=cases[`${format}-${language}`],c=await b.newContext({viewport:{width,height},hasTouch:format!=='desktop',isMobile:format==='mobile'});await c.addCookies(s.cookies);await c.addCookies([{name:'vibecore-lang',value:language,url:'http://127.0.0.1:5197'}]);const p=await c.newPage();let restores=0;p.on('request',r=>{if(r.method()==='POST' && /intent=restore(?:&|$)/.test(r.postData()??'')) restores++});await p.goto(`http://127.0.0.1:5197/projects/${s.projectId}/snapshots?lang=${language}`);const opener=p.getByRole('button',{name:/^(Restore|Restaurer) Focus /}).first();await opener.waitFor();const label=await opener.getAttribute('aria-label');await opener.focus();await p.keyboard.press('Enter');const dialog=p.getByRole('dialog',{name:/^(Restore|Restaurer)/});await expect(dialog).toBeVisible();const cancel=dialog.getByRole('button',{name:/^(Cancel|Annuler)$/});await expect(cancel).toBeFocused();await p.keyboard.press('Shift+Tab');await expect(dialog.getByRole('button').last()).toBeFocused();await p.keyboard.press('Tab');await expect(cancel).toBeFocused();await p.screenshot({path:`docs/bugs/ux-confirmation-2026-10-21/after-dialog-${format}-${language}.png`});if(format==='desktop')await p.keyboard.press('Enter');else await cancel.tap();await expect(dialog).toBeHidden();await expect(opener).toBeFocused();await p.screenshot({path:`docs/bugs/ux-confirmation-2026-10-21/after-cancel-${format}-${language}.png`});await p.keyboard.press('Enter');await expect(dialog).toBeVisible();await p.keyboard.press('Escape');await expect(dialog).toBeHidden();await expect(opener).toBeFocused();expect(restores).toBe(0);expect(await p.evaluate(()=>document.documentElement.scrollWidth>innerWidth)).toBe(false);results.push({format,language,label,persistedSnapshotLoaded:true,focusTrap:true,cancelRestoresFocus:true,escapeRestoresFocus:true,restoreRequests:restores,horizontalOverflow:false});await c.close();}
fs.writeFileSync('docs/bugs/ux-confirmation-2026-10-21/confirmation-existing.json',JSON.stringify(results,null,2));console.log(JSON.stringify(results,null,2));await b.close();})();

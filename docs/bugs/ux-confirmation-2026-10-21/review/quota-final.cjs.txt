const {chromium,expect}=require('@playwright/test');const fs=require('node:fs');
(async()=>{const s=JSON.parse(fs.readFileSync('.vibecore/ux-qa/snapshot-review-sessions.json')).cases['mobile-fr'];const b=await chromium.launch({executablePath:'/usr/bin/chromium'});const results=[];
for(const lang of ['en','fr']){const c=await b.newContext({viewport:{width:1440,height:900}});await c.addCookies(s.cookies);const p=await c.newPage();await p.goto(`http://127.0.0.1:5203/projects/${s.projectId}/snapshots?lang=${lang}`);await p.locator('#snapshot-label').fill(`Quota verification ${lang}`);const restore=p.getByRole('button',{name:lang==='fr'?/^Restaurer /:/^Restore /});let before=await restore.count();if(lang==='en'){for(let n=before;n<5;n++){const label=`Quota fixture ${Date.now()}`;await p.locator('#snapshot-label').fill(label);await p.getByRole('button',{name:'Create snapshot',exact:true}).click();await expect(p.getByRole('button',{name:`Restore ${label}`,exact:true})).toBeVisible();await expect(restore).toHaveCount(n+1);}before=await restore.count();await p.locator('#snapshot-label').fill(`Quota verification ${lang}`);}expect(before).toBe(5);const response=p.waitForResponse(r=>r.request().method()==='POST'&&new URL(r.url()).pathname.startsWith(`/projects/${s.projectId}/snapshots`));await p.getByRole('button',{name:lang==='fr'?'Créer l’instantané':'Create snapshot',exact:true}).click();const r=await response;expect(r.status()).toBe(429);expect(await r.text()).toContain('quotaExceeded');await expect(p.getByText(lang==='fr'?'Le quota d’instantanés de cet espace est atteint. Consultez l’utilisation et les limites de l’offre.':'The snapshot quota for this workspace has been reached. Check usage and plan limits.',{exact:true})).toBeVisible();expect(await restore.count()).toBe(before);await p.screenshot({path:`/workspace/ux-artifacts/quota-real-${lang}.png`});results.push({lang,status:r.status(),message:'quotaExceeded',snapshotsBefore:before,snapshotsAfter:await restore.count()});await c.close();}await b.close();fs.writeFileSync('/workspace/ux-artifacts/quota-real.json',JSON.stringify(results,null,2));console.log(JSON.stringify(results));})();

const{chromium,expect}=require('@playwright/test');const fs=require('node:fs');
(async()=>{const s=JSON.parse(fs.readFileSync('.vibecore/ux-qa/snapshot-review-sessions.json')).cases['mobile-fr'];const b=await chromium.launch({executablePath:'/usr/bin/chromium'});const c=await b.newContext({viewport:{width:390,height:320},hasTouch:true,isMobile:true});await c.addCookies(s.cookies);const p=await c.newPage();const prefs=p.waitForResponse(r=>r.url().includes('/api/user/preferences')&&r.request().method()==='GET');await p.goto(`http://127.0.0.1:5203/projects/${s.projectId}/snapshots?lang=fr`);await prefs;await p.getByRole('button',{name:'Aide et visite guidée',exact:true}).tap();await p.getByRole('button',{name:'Lancer la visite guidée',exact:true}).tap();const tour=p.locator('.vc-product-tour');await expect(tour).toBeVisible();const metrics=await tour.evaluate(e=>({height:e.clientHeight,scrollHeight:e.scrollHeight}));expect(metrics.scrollHeight).toBeGreaterThan(metrics.height);const box=await tour.boundingBox();const client=await c.newCDPSession(p);const x=box.x+30,start=box.y+box.height-50,end=box.y+30;await client.send('Input.dispatchTouchEvent',{type:'touchStart',touchPoints:[{x,y:start}]});for(let y=start-20;y>end;y-=20)await client.send('Input.dispatchTouchEvent',{type:'touchMove',touchPoints:[{x,y}]});await client.send('Input.dispatchTouchEvent',{type:'touchEnd',touchPoints:[]});await expect.poll(()=>tour.evaluate(e=>e.scrollTop)).toBeGreaterThan(0);const scrollTop=await tour.evaluate(e=>e.scrollTop);await p.screenshot({path:'/workspace/ux-artifacts/tour-scroll-mobile-fr.png'});const saved=p.waitForResponse(r=>r.url().includes('/api/user/preferences')&&r.request().method()==='PATCH');await tour.getByRole('button',{name:'Plus tard',exact:true}).tap();expect((await saved).ok()).toBe(true);await expect(tour).toBeHidden();console.log(JSON.stringify({viewport:[390,320],...metrics,scrollTop,touchScroll:true,laterReached:true}));await b.close()})();
